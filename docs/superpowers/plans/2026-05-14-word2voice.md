# Word2Voice LAN Android App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a standalone Android app in `phone_voice/` that exposes `POST /api/v1/word2voice` on the local network and speaks the incoming JSON `text` value through Android `TextToSpeech`.

**Architecture:** The app is a small Compose-based Android project with a foreground service that owns a `NanoHTTPD` server and a `TextToSpeech` wrapper. Request parsing, validation, state management, and runtime coordination are kept in focused Kotlin classes so the endpoint contract can be covered by unit tests before the Android service wiring is added.

**Tech Stack:** Kotlin, Jetpack Compose, Android foreground service APIs, `NanoHTTPD`, Android `TextToSpeech`, JUnit 4, Gradle Kotlin DSL

// phone_voice/app/src/main/java/com/example/phonevoice/http/WordToVoiceRequestHandler.kt
package com.example.phonevoice.http

import com.example.phonevoice.config.VoiceEndpointConfig
import org.json.JSONException
import org.json.JSONObject

class WordToVoiceRequestHandler(
    private val dispatcher: SpeechDispatcher,
    private val onAcceptedText: (String) -> Unit = {}
) {
    fun handle(method: String, path: String, body: String?): HttpResult {
        if (path != VoiceEndpointConfig.PATH) {
            return error(404, "path not found")
        }

        if (method != "POST") {
            return error(405, "method not allowed")
        }

        if (!dispatcher.isReady()) {
            return error(503, "speech engine not ready")
        }

        val payload = try {
            JSONObject(body.orEmpty())
        } catch (_: JSONException) {
            return error(400, "invalid json body")
        }

        val text = payload.optString("text", "").trim()
        if (text.isEmpty()) {
            return error(400, "text must not be blank")
        }

        if (text.length > VoiceEndpointConfig.MAX_TEXT_LENGTH) {
            return error(400, "text exceeds max length")
        }

        if (!dispatcher.speak(text)) {
            return error(500, "failed to speak text")
        }

        onAcceptedText(text)
        return HttpResult(
            statusCode = 200,
            body = JSONObject(
                mapOf(
                    "success" to true,
                    "message" to "accepted",
                    "text" to text
                )
            ).toString()
        )
    }

    private fun error(statusCode: Int, message: String): HttpResult {
        return HttpResult(
            statusCode = statusCode,
            body = JSONObject(
                mapOf(
                    "success" to false,
                    "message" to message
                )
            ).toString()
        )
    }
}

package com.example.phonevoice.state

import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update

class VoiceServiceStateStore {
    private val _state = MutableStateFlow(VoiceUiState())
    val state: StateFlow<VoiceUiState> = _state.asStateFlow()

    fun setStarting() {
        _state.update { it.copy(status = VoiceServiceStatus.STARTING, lastError = "") }
    }

    fun setRunning(lanUrl: String, ttsReady: Boolean) {

package com.example.phonevoice.net

import java.net.Inet4Address
import java.net.InetAddress
import java.net.NetworkInterface

object LocalIpResolver {
    fun resolveLanIpv4(): String? {
        val interfaces = NetworkInterface.getNetworkInterfaces() ?: return null
        val addresses = mutableListOf<InetAddress>()
        while (interfaces.hasMoreElements()) {
            val networkInterface = interfaces.nextElement()
            val interfaceAddresses = networkInterface.inetAddresses
            while (interfaceAddresses.hasMoreElements()) {
                addresses += interfaceAddresses.nextElement()
            }
        }
        return pickLanIpv4(addresses)
    }

    fun pickLanIpv4(addresses: List<InetAddress>): String? {
        return addresses
            .filterIsInstance<Inet4Address>()
            .firstOrNull { !it.isLoopbackAddress && it.isSiteLocalAddress }
            ?.hostAddress
    }
}

package com.example.phonevoice.runtime

import com.example.phonevoice.config.VoiceEndpointConfig
import com.example.phonevoice.http.SpeechDispatcher
import com.example.phonevoice.state.VoiceServiceStateStore

class VoiceRuntimeController(
    private val stateStore: VoiceServiceStateStore,
    private val dispatcher: SpeechDispatcher,
    private val lanIpProvider: () -> String?,
    private val serverFactory: () -> EmbeddedServer
) {
    private var server: EmbeddedServer? = null

    fun start() {
        if (server != null) {
            stateStore.setRunning(
                lanUrl = VoiceEndpointConfig.buildLanUrl(lanIpProvider()),
                ttsReady = dispatcher.isReady()
            )
            return
        }
        stateStore.setStarting()
        try {
            val createdServer = serverFactory()
            createdServer.start()
            server = createdServer
            stateStore.setRunning(
                lanUrl = VoiceEndpointConfig.buildLanUrl(lanIpProvider()),
                ttsReady = dispatcher.isReady()
            )
        } catch (t: Throwable) {
            stateStore.setError("Failed to start HTTP server: ${t.message ?: "unknown error"}")
        }
    }

    fun refreshTtsReady() {
        stateStore.setTtsReady(dispatcher.isReady())
    }

    fun stop() {
        server?.stop()
        server = null
        stateStore.setStopped()
    }
}

// phone_voice/app/src/main/java/com/example/phonevoice/http/WordToVoiceServer.kt
package com.example.phonevoice.http

import com.example.phonevoice.config.VoiceEndpointConfig
import com.example.phonevoice.runtime.EmbeddedServer
import fi.iki.elonen.NanoHTTPD
import java.io.IOException

class WordToVoiceServer(
    private val handler: WordToVoiceRequestHandler
) : NanoHTTPD(VoiceEndpointConfig.PORT), EmbeddedServer {

    override fun start() {
        super.start(SOCKET_READ_TIMEOUT, false)
    }

    override fun serve(session: IHTTPSession): Response {
        val body = readBody(session)
        return serveBody(session.method.name, session.uri, body)
    }

    fun serveBody(method: String, path: String, body: String?): Response {
        val result = handler.handle(method = method, path = path, body = body)
        return newFixedLengthResponse(
            statusFrom(result.statusCode),
            "application/json; charset=utf-8",
            result.body
        )
    }

    private fun readBody(session: IHTTPSession): String? {
        if (session.method != Method.POST) {
            return null
        }
        return try {
            val files = mutableMapOf<String, String>()
            session.parseBody(files)
            files["postData"]
        } catch (_: IOException) {
            null
        } catch (_: ResponseException) {
            null
        }
    }

    private fun statusFrom(code: Int): Response.IStatus {
        return when (code) {
            200 -> Response.Status.OK
            400 -> Response.Status.BAD_REQUEST
            404 -> Response.Status.NOT_FOUND
            405 -> Response.Status.METHOD_NOT_ALLOWED
            500 -> Response.Status.INTERNAL_ERROR
            503 -> Response.Status.SERVICE_UNAVAILABLE
            else -> Response.Status.INTERNAL_ERROR
        }
    }
}

package com.example.phonevoice.tts

import android.content.Context
import android.speech.tts.TextToSpeech
import com.example.phonevoice.http.SpeechDispatcher
import java.util.Locale
import java.util.UUID

class TtsManager(
    context: Context
) : SpeechDispatcher, TextToSpeech.OnInitListener {
    private val appContext = context.applicationContext
    private var tts: TextToSpeech? = null
    @Volatile private var ready = false

    fun initialize() {
        if (tts == null) {
            tts = TextToSpeech(appContext, this)
        }
    }

    override fun onInit(status: Int) {
        val engine = tts ?: return
        if (status != TextToSpeech.SUCCESS) {
            ready = false
            return
        }

        val languageResult = engine.setLanguage(Locale.SIMPLIFIED_CHINESE)
        ready = languageResult != TextToSpeech.LANG_MISSING_DATA &&
            languageResult != TextToSpeech.LANG_NOT_SUPPORTED
    }

    override fun isReady(): Boolean = ready

    override fun speak(text: String): Boolean {
        val engine = tts ?: return false
        val result = engine.speak(
            text,
            TextToSpeech.QUEUE_FLUSH,
            null,
            UUID.randomUUID().toString()
        )
        return result != TextToSpeech.ERROR
    }

    fun shutdown() {
        tts?.stop()
        tts?.shutdown()
        tts = null
        ready = false
    }
}

        super.onCreate()
        createNotificationChannel()
        ttsManager = TtsManager(this)

            dispatcher = ttsManager,

        when (intent?.action) {
            ACTION_STOP -> {
                controller.stop()
                stopForeground(STOP_FOREGROUND_REMOVE)
                stopSelf()
            }
            else -> {
                startForeground(NOTIFICATION_ID, buildNotification())
                controller.start()
                controller.refreshTtsReady()
            }
        }
        return START_STICKY
    }

        controller.stop()
        ttsManager.shutdown()
        super.onDestroy()
    }

    override fun onBind(intent: Intent?): IBinder? = null

        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setSmallIcon(android.R.drawable.ic_lock_silent_mode_off)
            .setContentTitle(getString(R.string.notification_title))
            .setContentText(getString(R.string.notification_text))
            .setOngoing(true)
            .build()
    }

        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.O) {
            return
        }

        manager.createNotificationChannel(

                CHANNEL_ID,
                getString(R.string.notification_channel_name),
                NotificationManager.IMPORTANCE_LOW
            )
        )

        private const val ACTION_STOP = "com.example.phonevoice.action.STOP"
